FROM python:3.10-slim

# Install system dependencies (FFmpeg and Node.js for JS extraction)
RUN apt-get update && apt-get install -y ffmpeg nodejs && rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .

# Install packages and force pre-release/nightly builds for yt-dlp
RUN pip install --no-cache-dir --upgrade --pre -r requirements.txt

COPY . .

CMD ["gunicorn", "-b", "0.0.0.0:10000", "--timeout", "180", "app:app"]
